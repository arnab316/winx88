import { useState, useEffect } from 'react';
import { ChevronLeft, ChevronRight, X, Info, ChevronRight as ChevronRightIcon, ChevronDown, Search, Pencil, Trash2, AlertCircle } from 'lucide-react';

import { promotionApi } from '../../api/promotionApi';
import type { Promotion as ApiPromotion } from '../../api/promotionApi';

export default function PromotionSettings() {
  const [filteredPromotions, setFilteredPromotions] = useState<ApiPromotion[]>([]);
  const [selectedPromotion, setSelectedPromotion] = useState<ApiPromotion | null>(null);
  const [isEditModalOpen, setIsEditModalOpen] = useState(false);
  const [, setIsLoading] = useState(false);
  const [totalRecords, setTotalRecords] = useState(0);

  const [deleteModalOpen, setDeleteModalOpen] = useState(false);
  const [promotionToDelete, setPromotionToDelete] = useState<ApiPromotion | null>(null);
  const [hardDelete, setHardDelete] = useState(false);
  const [deleteError, setDeleteError] = useState<string | null>(null);

  const [filters, setFilters] = useState({
    currency: 'BDT',
    status: '',
    bonusType: '',
    promoCode: ''
  });

  const [page, setPage] = useState(1);
  const itemsPerPage = 5;

  const fetchPromotions = async () => {
    setIsLoading(true);
    try {
      const res = await promotionApi.listPromotions({
        currency: filters.currency || undefined,
        isActive: filters.status ? filters.status === 'active' : undefined,
        kind: filters.bonusType || undefined,
        code: filters.promoCode || undefined,
        page,
        limit: itemsPerPage
      });
      setFilteredPromotions(res.data);
      setTotalRecords(res.total || 0);
    } catch (error) {
      console.error('Failed to fetch promotions:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleDeleteClick = (promo: ApiPromotion) => {
    setPromotionToDelete(promo);
    setHardDelete(false);
    setDeleteError(null);
    setDeleteModalOpen(true);
  };

  const confirmDelete = async () => {
    if (!promotionToDelete) return;
    setIsLoading(true);
    setDeleteError(null);
    try {
      await promotionApi.deletePromotion(promotionToDelete.id, hardDelete);
      setDeleteModalOpen(false);
      fetchPromotions();
    } catch (err: any) {
      setDeleteError(err.response?.data?.message || err.message || 'Failed to delete promotion');
    } finally {
      setIsLoading(false);
    }
  };

  const handleClearFilters = () => {
    setFilters({
      currency: 'BDT',
      status: '',
      bonusType: '',
      promoCode: ''
    });
  };

  const handleSearch = () => {
    setPage(1);
    fetchPromotions();
  };

  useEffect(() => {
    fetchPromotions();
  }, [page]);

  const handleCreate = () => {
    setSelectedPromotion({
      id: 0,
      title: '',
      kind: 'DEPOSIT',
      bonusType: 'FLAT',
      bonusValue: 0,
      isActive: true,
      currency: 'BDT',
    } as ApiPromotion);
    setIsEditModalOpen(true);
  };

  const handleEdit = (promotion: ApiPromotion) => {
    setSelectedPromotion(promotion);
    setIsEditModalOpen(true);
  };

  const handleSave = async (updatedPromo: ApiPromotion) => {
    try {
      if (updatedPromo.id === 0) {
        const { id, ...createPayload } = updatedPromo;
        if (!createPayload.code) {
          delete createPayload.code;
        }
        await promotionApi.createPromotion(createPayload);
      } else {
        const { id, code, currency, bonusTo, ...updatePayload } = updatedPromo as any;
        await promotionApi.updatePromotion(id, updatePayload);
      }
      await fetchPromotions();
      setIsEditModalOpen(false);
    } catch (err) {
      console.error('Failed to update/create promotion:', err);
    }
  };

  const totalPages = Math.ceil(totalRecords / itemsPerPage) || 1;
  const paginatedPromotions = filteredPromotions; // already paginated from backend

  return (
    <div className="flex flex-col gap-6 text-foreground bg-background/95 min-h-screen">
      {/* Header */}
      <div className="flex flex-col gap-1.5 border-b border-border pb-4">
        <h1 className="text-2xl font-bold tracking-tight">Promotion</h1>
        <p className="text-xs text-muted-foreground">Promotion Settings</p>
      </div>

      {/* Filters */}
      <div className="bg-card text-card-foreground rounded-xl border border-border p-6 shadow-sm">
        <h2 className="text-sm font-semibold mb-4 text-foreground">Promotion Settings</h2>

        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 mb-5">
          <div className="flex flex-col gap-1.5">
            <label className="text-xs font-semibold text-muted-foreground">Currency</label>
            <select
              value={filters.currency}
              onChange={(e) => setFilters({ ...filters, currency: e.target.value })}
              className="w-full px-3 py-1.5 text-xs border border-border rounded-md bg-background focus:border-primary focus:ring-1 focus:ring-primary/20 outline-none"
            >
              <option value="BDT">BDT</option>
              <option value="USD">USD</option>
              <option value="EUR">EUR</option>
            </select>
          </div>

          <div className="flex flex-col gap-1.5">
            <label className="text-xs font-semibold text-muted-foreground">Status</label>
            <select
              value={filters.status}
              onChange={(e) => setFilters({ ...filters, status: e.target.value })}
              className="w-full px-3 py-1.5 text-xs border border-border rounded-md bg-background focus:border-primary focus:ring-1 focus:ring-primary/20 outline-none"
            >
              <option value="">Please Select</option>
              <option value="active">Active</option>
              <option value="inactive">Inactive</option>
            </select>
          </div>

          <div className="flex flex-col gap-1.5">
            <label className="text-xs font-semibold text-muted-foreground">Bonus Type</label>
            <select
              value={filters.bonusType}
              onChange={(e) => setFilters({ ...filters, bonusType: e.target.value })}
              className="w-full px-3 py-1.5 text-xs border border-border rounded-md bg-background focus:border-primary focus:ring-1 focus:ring-primary/20 outline-none"
            >
              <option value="">Please Select</option>
              <option value="DEPOSIT">Deposit</option>
              <option value="REGISTRATION">Registration</option>
              <option value="FREE_REWARD">Free Reward</option>
              <option value="PROMOCODE">Promo Code</option>
              <option value="MANUAL">Manual</option>
              <option value="RELOAD">Reload</option>
              <option value="CASHBACK">Cashback</option>
              <option value="REBATE">Rebate</option>
            </select>
          </div>

          <div className="flex flex-col gap-1.5">
            <label className="text-xs font-semibold text-muted-foreground">Promo Code</label>
            <input
              type="text"
              value={filters.promoCode}
              onChange={(e) => setFilters({ ...filters, promoCode: e.target.value })}
              placeholder="Please Enter"
              className="w-full px-3 py-1.5 text-xs border border-border rounded-md bg-background focus:border-primary focus:ring-1 focus:ring-primary/20 outline-none"
            />
          </div>
        </div>

        <div className="flex justify-end gap-2.5">
          <button
            onClick={handleClearFilters}
            className="px-4 py-1.5 text-xs font-semibold border border-border rounded-md hover:bg-muted/50 transition-all active:scale-[0.98]"
          >
            Clear
          </button>
          <button 
            onClick={handleSearch}
            className="px-4 py-1.5 text-xs font-semibold bg-[#00a65a] hover:bg-[#008d4c] text-white rounded-md transition-all active:scale-[0.98]"
          >
            Search
          </button>
          <button 
            onClick={handleCreate}
            className="px-4 py-1.5 text-xs font-semibold bg-primary hover:bg-primary/90 text-primary-foreground rounded-md transition-all active:scale-[0.98]"
          >
            Create Promotion
          </button>
        </div>
      </div>

      {/* Table */}
      <div className="bg-card text-card-foreground rounded-xl border border-border shadow-sm overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-left border-collapse min-w-[1100px]">
            <thead>
              <tr className="border-b border-border bg-muted/10 text-xs font-semibold text-muted-foreground uppercase tracking-wider">
                <th className="py-3 px-4 w-12 text-center">ID</th>
                <th className="py-3 px-4">Currency</th>
                <th className="py-3 px-4">Member Group</th>
                <th className="py-3 px-4">Bonus Type</th>
                <th className="py-3 px-4">Calculation Type</th>
                <th className="py-3 px-4">Promo Code</th>
                <th className="py-3 px-4">Promo Name</th>
                <th className="py-3 px-4">Valid From</th>
                <th className="py-3 px-4">Valid Till</th>
                <th className="py-3 px-4 text-center">Status</th>
                <th className="py-3 px-4">Updated By</th>
                <th className="py-3 px-4">Updated Time</th>
                <th className="py-3 px-4 text-center">Action</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-border/60 text-xs text-foreground">
              {paginatedPromotions.map((promo) => {
                const isActive = promo.isActive ?? (promo as any).is_active ?? false;
                const memberGroupId = promo.memberGroupId || (promo as any).member_group_id || 'All';
                const targetType = promo.targetType || (promo as any).target_type || 'N/A';
                const startsAt = promo.startsAt || (promo as any).starts_at;
                const endsAt = promo.endsAt || (promo as any).ends_at;

                return (
                <tr key={promo.id} className="hover:bg-muted/10 transition-colors">
                  <td className="py-3.5 px-4 text-center font-mono text-muted-foreground">{promo.id}</td>
                  <td className="py-3.5 px-4">
                    <span className="rounded bg-fuchsia-500/10 px-2.5 py-0.5 text-[10px] font-bold text-fuchsia-600 dark:text-fuchsia-400 uppercase">
                      {promo.currency || 'ALL'}
                    </span>
                  </td>
                  <td className="py-3.5 px-4 font-medium">{memberGroupId}</td>
                  <td className="py-3.5 px-4 text-sky-600 dark:text-sky-400 font-semibold">{promo.kind || 'N/A'}</td>
                  <td className="py-3.5 px-4 text-muted-foreground">{targetType}</td>
                  <td className="py-3.5 px-4 font-mono font-medium">{promo.code || 'N/A'}</td>
                  <td className="py-3.5 px-4 max-w-[200px] truncate" title={promo.title}>{promo.title}</td>
                  <td className="py-3.5 px-4 text-muted-foreground whitespace-nowrap">{startsAt ? new Date(startsAt).toLocaleDateString() : 'N/A'}</td>
                  <td className="py-3.5 px-4 text-muted-foreground whitespace-nowrap">{endsAt ? new Date(endsAt).toLocaleDateString() : 'N/A'}</td>
                  <td className="py-3.5 px-4 text-center">
                    <span className={`inline-block rounded-full px-2.5 py-0.5 text-[10px] font-bold capitalize ${
                      isActive
                        ? 'bg-emerald-500/15 text-emerald-600 dark:text-emerald-400'
                        : 'bg-rose-500/15 text-rose-600 dark:text-rose-400'
                    }`}>
                      {isActive ? 'active' : 'inactive'}
                    </span>
                  </td>
                  <td className="py-3.5 px-4 text-muted-foreground">Admin</td>
                  <td className="py-3.5 px-4 text-muted-foreground whitespace-nowrap">N/A</td>
                  <td className="py-3.5 px-4 text-center">
                    <div className="flex items-center justify-center gap-1.5">
                      <button
                        onClick={() => handleEdit(promo)}
                        className="p-1.5 rounded-full bg-[#00a65a]/10 hover:bg-[#00a65a]/20 text-[#00a65a] transition-all hover:scale-105 active:scale-95"
                      >
                        <Pencil className="w-3.5 h-3.5" />
                      </button>
                      <button
                        onClick={() => handleDeleteClick(promo)}
                        className="p-1.5 rounded-full bg-rose-500/10 hover:bg-rose-500/20 text-rose-600 transition-all hover:scale-105 active:scale-95"
                      >
                        <Trash2 className="w-3.5 h-3.5" />
                      </button>
                    </div>
                  </td>
                </tr>
                );
              })}
              {filteredPromotions.length === 0 && (
                <tr>
                  <td colSpan={13} className="py-8 text-center text-muted-foreground">
                    No records found matching the search criteria.
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>

        {/* Pagination */}
        {filteredPromotions.length > 0 && (
          <div className="flex items-center justify-between px-6 py-4 border-t border-border">
            <button
              onClick={() => setPage(p => Math.max(1, p - 1))}
              disabled={page === 1}
              className="p-2 rounded-full border border-border hover:bg-muted text-foreground disabled:opacity-50 transition-colors"
            >
              <ChevronLeft className="w-4 h-4" />
            </button>
            <div className="text-xs text-muted-foreground font-semibold">
              Page {page} of {totalPages}
            </div>
            <button
              onClick={() => setPage(p => Math.min(totalPages, p + 1))}
              disabled={page === totalPages}
              className="p-2 rounded-full border border-border hover:bg-muted text-foreground disabled:opacity-50 transition-colors"
            >
              <ChevronRight className="w-4 h-4" />
            </button>
          </div>
        )}
      </div>

      {/* Modals */}
      {isEditModalOpen && selectedPromotion && (
        <EditPromotionModal
          promotion={selectedPromotion}
          onClose={() => setIsEditModalOpen(false)}
          onSave={handleSave}
        />
      )}

      {/* Delete Confirmation Modal */}
      {deleteModalOpen && promotionToDelete && (
        <div className="fixed inset-0 z-50 bg-background/80 backdrop-blur-sm flex items-center justify-center p-4 animate-in fade-in duration-200">
          <div className="bg-card w-full max-w-sm rounded-xl border border-border shadow-lg overflow-hidden animate-in zoom-in-95 duration-200">
            <div className="flex items-center justify-between px-4 py-3 border-b border-border bg-rose-500/10">
              <div className="flex items-center gap-2 text-rose-600 dark:text-rose-400">
                <AlertCircle className="w-4 h-4" />
                <h3 className="text-sm font-semibold">Delete Promotion</h3>
              </div>
              <button onClick={() => setDeleteModalOpen(false)} className="text-muted-foreground hover:text-foreground transition-colors p-1 hover:bg-muted rounded-md">
                <X className="w-4 h-4" />
              </button>
            </div>
            <div className="p-4 space-y-4">
              <p className="text-sm text-muted-foreground">
                Are you sure you want to delete the promotion <strong className="text-foreground">{promotionToDelete.code || promotionToDelete.title}</strong>?
              </p>
              
              {deleteError && (
                <div className="p-3 text-xs bg-rose-500/10 text-rose-600 border border-rose-500/20 rounded-md">
                  {deleteError}
                </div>
              )}
              
              <label className="flex items-center gap-2 mt-4 p-2 rounded border border-border bg-muted/30 cursor-pointer hover:bg-muted/50 transition-colors">
                <input
                  type="checkbox"
                  checked={hardDelete}
                  onChange={(e) => setHardDelete(e.target.checked)}
                  className="rounded border-border text-rose-600 size-3.5"
                />
                <span className="text-xs font-medium">Permanently delete (Hard Delete)</span>
              </label>
              <p className="text-[10px] text-muted-foreground px-1">
                Soft delete will deactivate the promotion. Hard delete permanently removes it but will fail if the promotion is actively referenced by claims or deposits.
              </p>
            </div>
            <div className="flex items-center justify-end gap-2 px-4 py-3 border-t border-border bg-muted/10">
              <button
                onClick={() => setDeleteModalOpen(false)}
                className="px-3 py-1.5 text-xs font-semibold border border-border rounded-md hover:bg-muted/50 transition-colors"
              >
                Cancel
              </button>
              <button
                onClick={confirmDelete}
                className="px-3 py-1.5 text-xs font-semibold bg-rose-600 text-white rounded-md hover:bg-rose-700 transition-colors"
              >
                Delete
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

interface EditPromotionModalProps {
  promotion: ApiPromotion;
  onClose: () => void;
  onSave: (updatedPromo: ApiPromotion) => void;
}

const gameProviders = ['LIVE', 'COCKFIGHT', 'FISH', 'SLOT', 'EGAMES', 'HORSE'];

export function EditPromotionModal({ promotion, onClose, onSave }: EditPromotionModalProps) {
  const [providerSearch, setProviderSearch] = useState('');
  const [expandedProviders, setExpandedProviders] = useState<Record<string, boolean>>({});
  const [selectedProviders, setSelectedProviders] = useState<Record<string, boolean>>({
    SLOT: true,
    EGAMES: true,
  });

  // Real member groups loaded from the backend (id + name). Falls back to the
  // known seed groups if the API isn't wired yet.
  const [memberGroups, setMemberGroups] = useState<{ id: number; name: string; code: string }[]>([]);
  useEffect(() => {
    (async () => {
      try {
        // Wire this to your API client: GET /member-group/admin
        const res = await promotionApi.listMemberGroups?.();
        if (res?.data?.length) {
          setMemberGroups(res.data.map((g: any) => ({ id: Number(g.id), name: g.name, code: g.code })));
          return;
        }
        throw new Error('no api');
      } catch {
        setMemberGroups([
          { id: 1, name: 'New Users', code: 'NEW_USERS' },
          { id: 2, name: 'All Members', code: 'ALL' },
          { id: 3, name: 'VIP Members', code: 'VIP' },
          { id: 4, name: 'Cricket Fans', code: 'CRICKET_FANS' },
        ]);
      }
    })();
  }, []);

  const toggleMemberGroup = (id: number) => {
    setFormData((prev) => {
      const ids = prev.memberGroupIds.includes(id)
        ? prev.memberGroupIds.filter((g) => g !== id)
        : [...prev.memberGroupIds, id];
      return { ...prev, memberGroupIds: ids };
    });
  };

  const [formData, setFormData] = useState({
    active: promotion.isActive ?? (promotion as any).is_active ?? false,
    currency: promotion.currency || 'BDT',
    deviceType: {
      desktop: (promotion.deviceTypes || (promotion as any).device_types)?.includes('DESKTOP') ?? true,
      mobile: (promotion.deviceTypes || (promotion as any).device_types)?.includes('MOBILE_WEB') ?? true,
      app: (promotion.deviceTypes || (promotion as any).device_types)?.includes('APP') ?? false
    },
    bonusCategory: promotion.kind || 'DEPOSIT',
    bonusName: {
      'en-bd': promotion.title || '',
      'bn-bd': ''
    },
    allowApply: true,
    code: promotion.code || '',
    linkedPromotion: promotion.linkedPromotionId ? String(promotion.linkedPromotionId) : '',
    validFrom: promotion.startsAt || (promotion as any).starts_at || '',
    validTill: promotion.endsAt || (promotion as any).ends_at || '',
    registerValidClaimablePeriod:
      (promotion as any).registerValidDays ?? (promotion as any).register_valid_days ?? '',
    registerValidClaimablePeriodNoMax:
      ((promotion as any).registerValidDays ?? (promotion as any).register_valid_days) == null,
    displayFrequency: 'Daily',
    rewardType: (promotion.bonusType || (promotion as any).bonus_type) === 'PERCENT' ? 'Percentage' : 'Fixed Amount',
    maxPlayer: promotion.maxPlayer ?? (promotion as any).max_player ?? '',
    maxPlayerNoMax: (promotion.maxPlayer ?? (promotion as any).max_player) == null,
    maxTotal: '',
    maxTotalNoMax: true,
    bonusAmount: promotion.bonusValue ?? (promotion as any).bonus_value ?? 0,
    maximumWithdrawal: promotion.maximumWithdrawal != null,
    balanceRequire: promotion.balanceRequire ?? (promotion as any).balance_require ?? 0,
    forfeitType: (promotion.forfeitType || (promotion as any).forfeit_type) === 'WALLET' ? 'Wallet Amount' : 'Bonus Amount',
    applyAmountMin: promotion.minAmount ?? (promotion as any).min_amount ?? 0,
    allowFlexibleApplyAmount: false,
    amountCap: promotion.maxBonus ?? (promotion as any).max_bonus ?? 0,
    amountCapNoMax: (promotion.maxBonus ?? (promotion as any).max_bonus) == null,
    removeMaxWithdrawLock: promotion.removeMaxWithdrawLock ?? (promotion as any).remove_max_withdraw_lock ?? false,
    lastDeposit: true,
    // Bonus Amount Provider Settings
    excludeGameList: 'Please Select',
    limitAccessToProvider: promotion.limitToProvider ?? (promotion as any).limit_to_provider ?? false,
    limitDeposit: false,
    autoApprove: promotion.autoApprove ?? (promotion as any).auto_approve ?? true,
    autoComplete: promotion.autoComplete ?? (promotion as any).auto_complete ?? false,
    autoUnlock: (promotion.autoUnlockThreshold ?? (promotion as any).auto_unlock_threshold) != null,
    autoUnlockBalance: promotion.autoUnlockThreshold ?? (promotion as any).auto_unlock_threshold ?? 1,
    allowCancel: promotion.allowCancel ?? (promotion as any).allow_cancel ?? false,
    allowCancelBalance: promotion.cancelThreshold ?? (promotion as any).cancel_threshold ?? '',
    allowCancelBalanceNoMax: (promotion.cancelThreshold ?? (promotion as any).cancel_threshold) == null,
    checkByWalletBalance: promotion.checkByWalletBalance ?? (promotion as any).check_by_wallet_balance ?? false,
    // Target Amount
    targetType: promotion.targetType || (promotion as any).target_type || 'TURNOVER',
    targetMultiplier: promotion.rolloverMultiplier ?? (promotion as any).rollover_multiplier ?? 0,
    targetOption: (promotion.targetOption || (promotion as any).target_option) === 'BONUS_ONLY' ? 'Bonus Only' : (promotion.targetOption || (promotion as any).target_option) === 'APPLY_ONLY' ? 'Apply Amount Only' : 'Bonus + Apply Amount',
    capLimitType: (promotion.capLimitType || (promotion as any).cap_limit_type) === 'PERCENT' ? 'Percentage' : 'Amount',
    // Eligibility — multiple member groups (ids). Reads memberGroups[] from the
    // API, falling back to the legacy single memberGroupId.
    promotionEligibleTypes:
      ((promotion as any).member_groups?.length || promotion.memberGroupId || (promotion as any).member_group_id)
        ? 'Member Groups'
        : 'All',
    memberGroupIds: (
      (promotion as any).member_groups?.map((g: any) => Number(g.id)) ??
      ((promotion.memberGroupId || (promotion as any).member_group_id)
        ? [Number(promotion.memberGroupId || (promotion as any).member_group_id)]
        : [])
    ) as number[],
    maxPerPlayer: promotion.maxUsesPerUser ?? (promotion as any).max_uses_per_user ?? '',
    maxPerPlayerNoMax: (promotion.maxUsesPerUser ?? (promotion as any).max_uses_per_user) == null,
    totalClaimTimes: promotion.maxUsesGlobal ?? (promotion as any).max_uses_global ?? '',
    totalClaimTimesNoMax: (promotion.maxUsesGlobal ?? (promotion as any).max_uses_global) == null,
    recurring: (promotion.frequency || (promotion as any).frequency) === 'ONE_TIME' ? 'None' : 'Daily',
    payLater: promotion.payLater ?? (promotion as any).pay_later ?? false,
    displayIfNonEligible: promotion.displayIfNonEligible ?? (promotion as any).display_if_non_eligible ?? false,
    hideIfEligible: promotion.hideIfEligible ?? (promotion as any).hide_if_eligible ?? false,
    // KYC Settings
    uniqueBankAccount: promotion.uniqueCheckBankAccount ?? (promotion as any).unique_check_bank_account ?? false,
    uniqueContactNo: promotion.uniqueCheckPhone ?? (promotion as any).unique_check_phone ?? false,
    uniqueEmail: promotion.uniqueCheckEmail ?? (promotion as any).unique_check_email ?? false,
    uniqueFingerprint: promotion.uniqueCheckDeviceFp ?? (promotion as any).unique_check_device_fp ?? false,
    uniqueIP: promotion.uniqueCheckIpAddress ?? (promotion as any).unique_check_ip_address ?? false,
    uniqueName: false,
    uniquePassword: false,
    violateCount: 0,
    verifyBank: false,
    verifyEmail: promotion.requireEmailVerified ?? (promotion as any).require_email_verified ?? false,
    verifyMobile: promotion.requirePhoneVerified ?? (promotion as any).require_phone_verified ?? false,
    verifyProfile: promotion.requireProfileVerified ?? (promotion as any).require_profile_verified ?? false,
    verifyCount: 1
  });

  const handleSave = () => {
    const deviceTypes: string[] = [];
    if (formData.deviceType.desktop) deviceTypes.push('DESKTOP');
    if (formData.deviceType.mobile) deviceTypes.push('MOBILE_WEB');
    if (formData.deviceType.app) deviceTypes.push('APP');

    onSave({
      id: promotion.id,
      isActive: formData.active,
      currency: formData.currency,
      kind: formData.bonusCategory as any,
