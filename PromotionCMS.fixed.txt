  const CURRENCY_TZ: Record<string, string> = {
    BDT: 'Asia/Dhaka',     // UTC+6
    INR: 'Asia/Kolkata',   // UTC+5:30
    PKR: 'Asia/Karachi',
    NPR: 'Asia/Kathmandu',
    USD: 'UTC',
  };

  function toUtcIso(localDatetime?: string, currency: string = 'BDT') {
    if (!localDatetime) return undefined;
    const tz = CURRENCY_TZ[currency] || 'UTC';
    return dayjs.tz(localDatetime, tz).utc().toISOString();
  }

  function fromUtcIso(iso?: string, currency: string = 'BDT') {
    if (!iso) return '';
    const tz = CURRENCY_TZ[currency] || 'UTC';
    return dayjs.utc(iso).tz(tz).format('YYYY-MM-DDTHH:mm');
  }
